实验三（参考实验指导书实验4-8）：数控分频器设计
参考程序：
module DVF(CLK, D, FOUT);   //数控分频器实验源程序
   input       CLK;    //实验箱下载时，CLK选用32768Hz
   input [7:0] D;
   output      FOUT;
   reg         FOUT;
   reg         FULL;
   reg [7:0]   CNT8;
   reg         CNT2;            
   
	always @(posedge CLK)
      begin
		   if (CNT8 == 8'b11111111)
           begin
            CNT8 = D;        //当CNT8计数计满时，输入数据D被同步预置给计数器CNT8 
            FULL <= 1'b1;    //同时使溢出标志信号FULL输出为高电平 
           end 
         else
           begin 
            CNT8 = CNT8 + 1;   //否则继续作加1计数
            FULL <= 1'b0;      //且输出溢出标志信号FULL为低电平       
           end
      end		
	
   always @(posedge FULL)
     begin
         CNT2 = (~CNT2);   //如果溢出标志信号FULL为高电平，D触发器输出取反
         if (CNT2 == 1'b1)      
            FOUT <= 1'b1;
         else
            FOUT <= 1'b0;
      end
endmodule

// Design Name: 扩展板流水灯实验
// Module Name: liushuideng
module liushuideng(clk50MHz,led,rst);  
   input clk50MHz;
	input rst;
	output reg [11:0]  led;      //接P1IO口
	reg clk4hz;
	reg [24:0] count;
	
	always@(posedge clk50MHz )
		   if(count==25'd6250000)
	         begin   clk4hz = ~clk4hz;
	                 count =25'd0;
　　　　　　　end
	       else count =count + 1;  
	always@(posedge clk4hz or negedge rst)
	  if(!rst)  
	        led=12'hffe;          //12'hffc=12'b111111100000;led灯为共阳极，低电平点亮。
		else   led = { led[10:0],led[11] };
endmodule

//clkdiv子模块参考程序：50MHz的输入时钟分频为1Hz
module clkdiv(clkin,clk1Hz);  
input clkin;   //输入时钟频率50MHz
output clk1Hz; //分频结果为1Hz
reg clk1Hz;
reg [25:0]cnt1;  //50M分频所用计数器
reg full;     //计数器溢出标志信号
reg cnt2;     //1Hz中间变量

always@(posedge clkin)
  if(cnt1>=26'd24999999)  //计数器记到25M归零，得到半个周期
    begin
      full<=1'b1;
      cnt1<=26'd0;
    end 
  else  begin 
    cnt1<=cnt1+1'b1; 
	 full<=1'b0;
	 end
 always @ (posedge full) 
    begin 
	 cnt2<=~cnt2;
	 clk1Hz=cnt2;
	 end
endmodule 

（仿真测试有两种办法，方法一是按上述源程序仿真，因计数器每次要计到24999999，仿真总时间要设置100us，clk要设置0.01ns,仿真运行时间十分钟以上，才能得到比较完整结果；方法二是把24999999改为249，只验证程序是否有问题，用500分频即可，仿真时间短，容易看到仿真效果，仿真结束后再改回24999999并再次编译，推荐用方法二仿真）

//traffic子模块
module traffic(clk, r1, y1, g1, r2, y2, g2);
   input      clk;
   output     r1;
   reg        r1;
   output     y1;
   reg        y1;
   output     g1;
   reg        g1;
   output     r2;
   reg        r2;
   output     y2;
   reg        y2;
   output     g2;
   reg        g2;
   parameter  yellow_time = 5;
   parameter  green_time = 20;
   reg [1:0]  state;
   reg [4:0]  count;
   always @(posedge clk)
      
         case (state)
            0 :
               begin
                  r1 <= 1'b0;
                  y1 <= 1'b0;
                  g1 <= 1'b1;
                  r2 <= 1'b1;
                  y2 <= 1'b0;
                  g2 <= 1'b0;
               end
            1 :
               begin
                  r1 <= 1'b0;
                  y1 <= 1'b1;
                  g1 <= 1'b0;
                  r2 <= 1'b1;
                  y2 <= 1'b0;
                  g2 <= 1'b0;
               end
            2 :
               begin
                  r1 <= 1'b1;
                  y1 <= 1'b0;
                  g1 <= 1'b0;
                  r2 <= 1'b0;
                  y2 <= 1'b0;
                  g2 <= 1'b1;
               end
            default :
               begin
                  r1 <= 1'b1;
                  y1 <= 1'b0;
                  g1 <= 1'b0;
                  r2 <= 1'b0;
                  y2 <= 1'b1;
                  g2 <= 1'b0;
               end
         endcase
   
   always @(posedge clk)       
      begin
         if (count == 5'd0)
         begin
            if (state == 2'd3)
               state <= 2'd0;
            else
               state <= state + 1'b1;
            case (state)
               2'b00 :
                  count <= yellow_time;
               2'b01 :
                  count <= green_time;
               2'b10 :
                  count <= yellow_time;
               default :
                  count <= green_time;
            endcase
         end
         else
            count = count - 1'b1;
      end   
endmodule
